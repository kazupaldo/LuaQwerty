-- Auto Steal Egg with WindUI-like controls
-- Add rarity, tween speed, and height speed settings in a WindUI-style panel

local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")
local TweenService = game:GetService("TweenService")

local localPlayer = Players.LocalPlayer
local config = {
    Enabled = true,
    TargetSpecies = "All",
    SearchRange = 200,
    StealDistance = 8,
    TweenSpeed = 6,
    HeightSpeed = 2.5,
    StealDelay = 0.2,
}

local areaEggRarityCache = {}

local rarityOrder = {
    Common = 1,
    Rare = 2,
    Epic = 3,
    Legendary = 4,
    Mythic = 5,
}

local function getRootPart()
    local char = localPlayer.Character
    if not char then
        return nil
    end
    return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("RootPart")
end

local function isEggLikeName(name)
    if not name then
        return false
    end
    local lower = string.lower(name)
    return (lower:find("egg") ~= nil) and (lower:find("fake") == nil)
end

local function getPartFromModel(model)
    if not model then
        return nil
    end

    local root = model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("RootPart") or model:FindFirstChild("PrimaryPart")
    if root and root:IsA("BasePart") then
        return root
    end

    for _, child in ipairs(model:GetChildren()) do
        if child:IsA("BasePart") then
            return child
        end
    end

    return nil
end

local detectedSpecies = {}

local function detectAllSpecies()
    detectedSpecies = {}
    local ok, result = pcall(function()
        local eggModels = game:GetService("ReplicatedStorage"):FindFirstChild("Assets")
        if eggModels then
            eggModels = eggModels:FindFirstChild("Models")
        end
        if eggModels then
            eggModels = eggModels:FindFirstChild("Eggs")
        end
        if not eggModels then
            return {}
        end

        local species = {}
        for _, child in ipairs(eggModels:GetChildren()) do
            local name = child.Name
            if name and name ~= "" then
                species[#species + 1] = name
            end
        end
        return species
    end)
    if ok and result then
        detectedSpecies = result
    end
    return detectedSpecies
end

local function getEggRarityFromText(input)
    if not input then
        return "Common"
    end

    local lowered = string.lower(input)
    for _, rarity in ipairs({"Mythic", "Legendary", "Epic", "Rare", "Common"}) do
        if lowered:find(string.lower(rarity), 1, true) then
            return rarity
        end
    end

    return "Common"
end

local function readRarityValue(instance)
    if not instance then
        return nil
    end

    local ok, result = pcall(function()
        local candidates = {}
        local success1, val1 = pcall(function() return instance.Rarity end)
        local success2, val2 = pcall(function() return instance.EggRarity end)
        local success3, val3 = pcall(function() return instance.rarity end)
        local success4, val4 = pcall(function() return instance.Type end)
        local success5, val5 = pcall(function() return instance.EggType end)

        if success1 and val1 then candidates[#candidates + 1] = val1 end
        if success2 and val2 then candidates[#candidates + 1] = val2 end
        if success3 and val3 then candidates[#candidates + 1] = val3 end
        if success4 and val4 then candidates[#candidates + 1] = val4 end
        if success5 and val5 then candidates[#candidates + 1] = val5 end

        for _, value in ipairs(candidates) do
            if value ~= nil then
                local text = tostring(value)
                if text ~= "" then
                    local rarity = getEggRarityFromText(text)
                    if rarity ~= "Common" or string.lower(text):find("common") or string.lower(text):find("rare") or string.lower(text):find("epic") or string.lower(text):find("legendary") or string.lower(text):find("mythic") then
                        return rarity
                    end
                end
            end
        end

        for _, child in ipairs(instance:GetChildren()) do
            local childName = string.lower(child.Name or "")
            local childValue = tostring(child.Value or "")
            if child:IsA("StringValue") or child:IsA("IntValue") or child:IsA("ObjectValue") then
                if childName:find("rarity") or childValue:find("rarity") or childName:find("type") or childValue:find("common") or childValue:find("rare") or childValue:find("epic") or childValue:find("legendary") or childValue:find("mythic") then
                    local rarity = getEggRarityFromText(childValue)
                    if rarity ~= "Common" or childName:find("rarity") then
                        return rarity
                    end
                end
            end
        end

        local attrSuccess1, attrVal1 = pcall(function() return instance:GetAttribute("Rarity") end)
        local attrSuccess2, attrVal2 = pcall(function() return instance:GetAttribute("EggRarity") end)
        local attrSuccess3, attrVal3 = pcall(function() return instance:GetAttribute("rarity") end)

        local attrRarity = (attrSuccess1 and attrVal1) or (attrSuccess2 and attrVal2) or (attrSuccess3 and attrVal3)
        if attrRarity then
            return getEggRarityFromText(tostring(attrRarity))
        end

        return nil
    end)

    return ok and result or nil
end

local function refreshAreaEggRarityCache()
    areaEggRarityCache = {}
    local root = Workspace:FindFirstChild("AreaEggSlotsClient")
    if not root then
        return
    end

    for _, slot in ipairs(root:GetDescendants()) do
        local rarity = readRarityValue(slot)
        if rarity then
            local nameLower = string.lower(slot.Name or "")
            local parentLower = slot.Parent and string.lower(slot.Parent.Name or "") or ""
            areaEggRarityCache[nameLower] = rarity
            if parentLower ~= "" then
                areaEggRarityCache[parentLower] = rarity
            end
        end
    end
end

local function getActualRarityForObject(obj)
    if not obj then
        return "Common"
    end

    local direct = readRarityValue(obj)
    if direct then
        return direct
    end

    local nameLower = string.lower(obj.Name or "")
    for key, rarity in pairs(areaEggRarityCache) do
        if key ~= "" and (nameLower:find(key, 1, true) or key:find(nameLower, 1, true)) then
            return rarity
        end
    end

    return getEggRarityFromText(obj.Name or "")
end

local function getEggRarity(obj)
    if not obj then
        return "Common"
    end

    return getActualRarityForObject(obj)
end

local function findEggs()
    refreshAreaEggRarityCache()

    local root = getRootPart()
    if not root then
        return {}
    end

    local results = {}

    for _, obj in ipairs(Workspace:GetDescendants()) do
        local name = obj.Name
        local isPart = obj:IsA("BasePart")
        local isModel = obj:IsA("Model")

        if (isPart and isEggLikeName(name)) or (isModel and isEggLikeName(name)) then
            local part = isPart and obj or getPartFromModel(obj)
            if part then
                local rarity = getEggRarity(obj)
                local objNameLower = string.lower(obj.Name or "")
                local matchesSpecies = config.TargetSpecies == "All"
                
                if not matchesSpecies then
                    for _, species in ipairs(detectedSpecies) do
                        if string.lower(species):find(objNameLower, 1, true) or objNameLower:find(string.lower(species), 1, true) then
                            matchesSpecies = true
                            break
                        end
                    end
                    if not matchesSpecies and objNameLower:find(string.lower(config.TargetSpecies), 1, true) then
                        matchesSpecies = true
                    end
                end
                
                if matchesSpecies then
                    table.insert(results, {
                        part = part,
                        distance = (part.Position - root.Position).Magnitude,
                        rarity = rarity,
                    })
                end
            end
        end
    end

    table.sort(results, function(a, b)
        return a.distance < b.distance
    end)

    return results
end

local function safeFirePrompt(prompt)
    if not prompt or not prompt.Enabled then
        return false
    end

    if type(fireproximityprompt) == "function" then
        fireproximityprompt(prompt)
        return true
    end

    local ok = pcall(function()
        prompt:InputHoldBegin()
    end)

    if ok then
        task.wait(0.05)
        pcall(function()
            prompt:InputHoldEnd()
        end)
        return true
    end

    return false
end

local function safeFireDetector(detector)
    if not detector then
        return false
    end

    if type(fireclickdetector) == "function" then
        fireclickdetector(detector)
        return true
    end

    return false
end

local function interactWithEgg(part)
    if not part then
        return false
    end

    local prompt = part:FindFirstChildOfClass("ProximityPrompt")
    local parent = part.Parent
    if not prompt and parent then
        prompt = parent:FindFirstChildOfClass("ProximityPrompt")
    end

    if prompt then
        return safeFirePrompt(prompt)
    end

    local detector = part:FindFirstChildOfClass("ClickDetector")
    if not detector and parent then
        detector = parent:FindFirstChildOfClass("ClickDetector")
    end

    if detector then
        return safeFireDetector(detector)
    end

    return false
end

local function moveToEgg(targetPos)
    local root = getRootPart()
    local hum = localPlayer.Character and localPlayer.Character:FindFirstChildOfClass("Humanoid")
    if not root or not hum then
        return
    end

    local finalPos = targetPos + Vector3.new(0, config.HeightSpeed * 0.75, 0)
    local distance = (finalPos - root.Position).Magnitude

    if distance > 1 then
        hum:MoveTo(finalPos)
        local waitTime = math.clamp(0.35 / math.max(config.TweenSpeed, 0.1), 0.04, 0.35)
        task.wait(waitTime)
    end
end

local function stealLoop()
    while config.Enabled do
        local root = getRootPart()
        if root then
            local nearest = nil
            local nearestDistance = math.huge

            for _, egg in ipairs(findEggs()) do
                if egg.distance <= config.SearchRange and egg.distance < nearestDistance then
                    nearest = egg.part
                    nearestDistance = egg.distance
                end
            end

            if nearest then
                local dist = (nearest.Position - root.Position).Magnitude

                if dist > config.StealDistance then
                    moveToEgg(nearest.Position)
                else
                    interactWithEgg(nearest)
                end
            end
        end

        task.wait(config.StealDelay)
    end
end
