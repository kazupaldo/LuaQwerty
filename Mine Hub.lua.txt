        local root = Game.getRoot()
        if not root then return false end

        local tweenInfo = TweenInfo.new(speed, Enum.EasingStyle.Linear)
        local tween = TweenService:Create(root, tweenInfo, { CFrame = targetCFrame })

        local root = Game.getRoot()
        if not root then return nil end

        local minDist = math.huge
        for _, model in pairs(GameWorkspace:GetChildren()) do
            if model:IsA("Model") then
                local dist = (model:GetPivot().Position - root.CFrame.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    nearest = model

            end

    end,

        local args = table.pack(...)
        local result
        local success = false
        task.spawn(function()
            local ok, ret = pcall(remote.InvokeServer, remote, table.unpack(args, 1, args.n))
            result = ret
            success = ok
        end)
        local timeout = os.clock() + 1
        while os.clock() < timeout and not success do task.wait() end
        if success then return result end
        return nil
    end,

    netRemote = function(remote, ...)
        remote:FireServer(...)
    end,

    httpPost = function(url, payload)
        local body = HttpService:JSONEncode(payload)
        local requestFunc = http_request or request or (syn and syn.request) or (http and http.request)
        if not requestFunc then return false end
        local response = requestFunc({
            Url = url,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = body
        })
        return response
    end,

-- STEAL FUNCTIONS
local function stealEgg(eggData)
    if not eggData then return false end
    local uid = eggData.uid or eggData["uid"]
    if typeof(uid) ~= "string" then return false end
    local eggCFrame = Game.getEggPosition(eggData)
    if not eggCFrame then return false end
    Game.tweenTo(eggCFrame)
    Remotes.RequestCarryAreaEgg:FireServer(uid)
    task.wait(0.3)
    Game.returnToBase()
    task.wait(0.2)
    Remotes.RequestDropHeldAreaEgg:FireServer()
    EggsStolen = 1
    return true
end

local function pickStealTarget()
    local eggs = Game.getAreaEggs()
    if typeof(eggs) ~= "table" then return nil end
    local bestEgg = nil
    local bestScore = -math.huge
    for uid, egg in pairs(eggs) do
        local rarity = tonumber(egg.rarity) or 0
        local size = egg.size or 0
        local score = rarity * 100 + size
        if score > bestScore then
            bestScore = score
            bestEgg = egg
        end
    end
    return bestEgg
end

local function runAutoSteal()
    if isStealing then return end
    isStealing = true
    while Toggles.AutoStealAll.Value do
        local target = pickStealTarget()
        if not target then
            if Toggles.AutoServerHop.Value and ServerHopMode == "No Matching Eggs" then
                serverHop()
            end
            break
        end
        stealEgg(target)
        task.wait(0.5)
    end
    isStealing = false
end

-- SERVER HOP
local function fetchServerPage(cursor)
    local url = string.format(
        "https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&excludeFullGames=true&limit=100",
        PlaceId
    )
    if cursor then url = url .. "&cursor=" .. cursor end
    local success, response = pcall(function() return game:HttpGet(url) end)
    if not success then return nil end
    return HttpService:JSONDecode(response)
end

local function serverHop()
    if isServerHopping then return end
    isServerHopping = true
    local cursor = nil
    local candidates = {}
    while true do
        local page = fetchServerPage(cursor)
        if typeof(page) ~= "table" then break end
        for _, server in pairs(page.data) do
            if server.id ~= JobId and server.playing < server.maxPlayers then
                table.insert(candidates, { id = server.id, playing = server.playing })
            end
        end
        if #candidates >= 2 then break end
        if page.nextPageCursor then cursor = page.nextPageCursor else break end
    end
    table.sort(candidates, function(a, b) return a.playing < b.playing end)
    if #candidates == 0 then
        UI.Notify("Server hop found no candidates, retrying in 30s")
        task.wait(30)
        isServerHopping = false
        return
    end
    local target = candidates[1]
    local success = pcall(function()
        TeleportService:TeleportToPlaceInstance(PlaceId, target.id, LocalPlayer)
    end)
    if not success then
        UI.Notify("Server hop failed, retrying in 10s")
        task.wait(10)
    end
    isServerHopping = false
end

-- FPS BOOST
local FpsBoostState = { enabled = false, savedSettings = {}, savedParts = {}, connection = nil }

local function applyFpsBoost(enabled)
    if enabled then
        local lighting = Lighting
        FpsBoostState.savedSettings = {
            GlobalShadows = lighting.GlobalShadows,
            FogEnd = lighting.FogEnd,
            Brightness = lighting.Brightness,
        }
        pcall(function() settings().Studio.WaitTime = Enum.Studio.WaitTime.Never end)
        lighting.GlobalShadows = false
        lighting.FogEnd = 9e9
        for _, part in pairs(workspace:GetDescendants()) do
            if part:IsA("BasePart") or part:IsA("Decal") or part:IsA("Texture") then
                table.insert(FpsBoostState.savedParts, part)
                part.CastShadow = false
            end
        end
        FpsBoostState.connection = workspace.DescendantAdded:Connect(function(part)
            if part:IsA("BasePart") or part:IsA("Decal") or part:IsA("Texture") then
                part.CastShadow = false
            end
        end)
        FpsBoostState.enabled = true
    else
        if FpsBoostState.savedSettings.GlobalShadows ~= nil then
            lighting.GlobalShadows = FpsBoostState.savedSettings.GlobalShadows
            lighting.FogEnd = FpsBoostState.savedSettings.FogEnd
            lighting.Brightness = FpsBoostState.savedSettings.Brightness
        end
        for _, part in pairs(FpsBoostState.savedParts) do
            if part and part.Parent then
                part.CastShadow = true
            end
        end
        if FpsBoostState.connection then
            FpsBoostState.connection:Disconnect()
            FpsBoostState.connection = nil
        end
        FpsBoostState.enabled = false
    end
end

-- FLY
local FlyState = { enabled = false, speed = 50, connection = nil }

local function toggleFly(enabled)
    local root = Game.getRoot()
    if not root then return end
    if enabled then
        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bv.Velocity = Vector3.new(0, 0, 0)
        bv.Parent = root
        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        bg.P = 10000
        bg.CFrame = root.CFrame
        bg.Parent = root
        FlyState.connection = RunService.RenderStepped:Connect(function()
            local character = LocalPlayer.Character
            if not character then return end
            local rootPart = character:FindFirstChild("HumanoidRootPart")
            if not rootPart then return end
            local camera = workspace.CurrentCamera
            local moveVector = Vector3.new(0, 0, 0)
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                moveVector = moveVector + camera.CFrame.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                moveVector = moveVector - camera.CFrame.LookVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                moveVector = moveVector - camera.CFrame.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                moveVector = moveVector + camera.CFrame.RightVector
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                moveVector = moveVector + Vector3.new(0, 1, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                moveVector = moveVector - Vector3.new(0, 1, 0)
            end
            bv.Velocity = moveVector * FlyState.speed
            bg.CFrame = camera.CFrame
        end)
        FlyState.enabled = true
    else
        if FlyState.connection then
            FlyState.connection:Disconnect()
            FlyState.connection = nil
        end
        local bv = root:FindFirstChild("BodyVelocity")
        if bv then bv:Destroy() end
        local bg = root:FindFirstChild("BodyGyro")
        if bg then bg:Destroy() end
        FlyState.enabled = false
    end
end

-- BUILD SUMMARY EMBED
local function buildSummaryEmbed()
    local elapsed = math.floor(os.clock() - SessionStartTime)
    local timeStr = string.format("%dh %dm",
    math.floor(elapsed / 3600),
    math.floor((elapsed % 3600) / 60)
)
    return {
        title = "PALDO | Session Summary",
        description = "**Player** `" .. LocalPlayer.Name .. "`\n**Runtime** `" .. timeStr .. "`",
        color = 0x7fd47f,
        fields = {
            { name = "Eggs stolen", value = "**" .. tostring(EggsStolen) .. "**", inline = true },
            { name = "Session time", value = timeStr, inline = true },
        },
        footer = { text = "Paldo Hub" },
        timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
    }
end

RunService.Heartbeat:Connect(function()
    -- Auto Steal
    if Toggles.AutoStealAll and Toggles.AutoStealAll.Value and not isStealing then
        task.spawn(runAutoSteal)
    end
    
    -- Auto Sell Pets
    if Toggles.AutoSellPets and Toggles.AutoSellPets.Value then
        task.spawn(runAutoSellPets)
    end
    
    -- Auto Fuse
    if Toggles.AutoFusePets and Toggles.AutoFusePets.Value then
        task.spawn(runAutoFusePets)
    end
    
    -- Auto Claim Index
    if Toggles.AutoClaimIndex and Toggles.AutoClaimIndex.Value then
        task.spawn(runAutoClaimIndex)
    end
    
    -- Auto Upgrades
    if Toggles.AutoUpgrades and Toggles.AutoUpgrades.Value then
        task.spawn(runAutoUpgrades)
    end
    
    -- Auto Open Ready Eggs
    if Toggles.AutoOpenReadyEggs and Toggles.AutoOpenReadyEggs.Value then
        task.spawn(runAutoOpenReadyEggs)
    end
    
    -- Auto Place Eggs
    if Toggles.AutoPlaceAll and Toggles.AutoPlaceAll.Value then
        task.spawn(runAutoPlaceEggs)
    end
    
    -- Auto Treadmill
    if Toggles.AutoTreadmill and Toggles.AutoTreadmill.Value and not isTreadmill then
        task.spawn(runAutoTreadmill)
    end
end)

-- INITIALIZATION
local function init()
    if getgenv().OuroborosLoaded then return end
    getgenv().OuroborosLoaded = true
    
    -- Load game remotes
    local clientModule = ReplicatedStorage:WaitForChild("Ouroboros"):WaitForChild("Client")
    Remotes = clientModule:WaitForChild("Remotes")
    Client = require(clientModule:WaitForChild("Client"))
    GameConfig = require(ReplicatedStorage:WaitForChild("Ouroboros"):WaitForChild("Config"))
    Areas = GameConfig.Areas
    Rarities = GameConfig.Rarities
    Mutations = GameConfig.Mutations
    
    UI.Notify("Paldo Hub loaded successfully!")
end

init()
